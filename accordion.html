NỘI DUNG JAVASCRIPT (DOM) 
// 1. Đóng/Mở toàn bộ Accordion
const accHeader = document.querySelector("#accHeader");
const accBody = document.querySelector("#accBody");
const accArrow = document.querySelector("#accArrow");

accHeader.addEventListener("click", function () {
  accBody.classList.toggle("hidden");
  accArrow.classList.toggle("rotate-180");
});

// 2. Đóng/Mở danh sách "Việc cần làm đã hoàn thành"
const btnCompletedToggle = document.querySelector("#btnCompletedToggle");
const completedList = document.querySelector("#completedList");
const completedArrow = document.querySelector("#completedArrow");

btnCompletedToggle.addEventListener("click", function () {
  completedList.classList.toggle("hidden");
  completedArrow.classList.toggle("rotate-180");
});

// 3. Đánh dấu hoàn thành nhiệm vụ & Cập nhật số đếm
const btnCheckTask = document.querySelector("#btnCheckTask");
const checkIcon = document.querySelector("#checkIcon");
const taskTitle = document.querySelector("#taskTitle");
const taskCount = document.querySelector("#taskCount");

let isDone = false;
btnCheckTask.addEventListener("click", function () {
  isDone = !isDone;
  if (isDone) {
    checkIcon.classList.remove("hidden");
    btnCheckTask.classList.add("border-blue-600", "bg-blue-50");
    taskTitle.classList.add("line-through", "text-slate-400");
    taskCount.innerText = "Còn 3 / 6 nhiệm vụ";
  } else {
    checkIcon.classList.add("hidden");
    btnCheckTask.classList.remove("border-blue-600", "bg-blue-50");
    taskTitle.classList.remove("line-through", "text-slate-400");
    taskCount.innerText = "Còn 4 / 6 nhiệm vụ";
  }
});

// 4. Tương tác Like / Dislike
const btnLike = document.querySelector("#btnLike");
const btnDislike = document.querySelector("#btnDislike");

btnLike.addEventListener("click", function () {
  btnLike.classList.toggle("text-blue-600");
  btnDislike.classList.remove("text-red-600");
});

btnDislike.addEventListener("click", function () {
  btnDislike.classList.toggle("text-red-600");
  btnLike.classList.remove("text-blue-600");
});
