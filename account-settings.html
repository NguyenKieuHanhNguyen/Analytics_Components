  row.addEventListener("click", function (e) {
    // Không kích hoạt nếu nhấn vào nút trợ giúp (?)
    if (e.target.closest(".group\\/tip")) return;

    const label = this.getAttribute("data-label");
    console.log(`Đang mở: ${label}`);
  });
});
