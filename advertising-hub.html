NỘI DUNG JAVASCRIPT (DOM) 
// 1. LOGIC CHUYỂN ĐỔI 2 TABS QUẢNG CÁO / NHÀ XUẤT BẢN
const tabAdsBtn = document.querySelector("#tabAdsBtn");
const tabPubBtn = document.querySelector("#tabPubBtn");
const tabDescText = document.querySelector("#tabDescText");
const productItems = document.querySelectorAll(".product-item");

tabAdsBtn?.addEventListener("click", () => {
  tabAdsBtn.className = "tab-btn px-4 py-3 font-medium border-b-2 border-blue-600 text-blue-600 cursor-pointer";
  tabPubBtn.className = "tab-btn px-4 py-3 font-medium border-b-2 border-transparent text-slate-600 hover:text-slate-900 cursor-pointer";
  tabDescText.innerText = "Để xem các lộ trình chuyển đổi hàng đầu, hiệu suất kênh và so sánh các mô hình phân bổ, hãy liên kết với một sản phẩm quảng cáo của Google.";
  
  productItems.forEach(item => {
    item.classList.toggle("hidden", item.getAttribute("data-type") !== "ads");
  });
  selectProduct("Google Ads");
});

tabPubBtn?.addEventListener("click", () => {
  tabPubBtn.className = "tab-btn px-4 py-3 font-medium border-b-2 border-blue-600 text-blue-600 cursor-pointer";
  tabAdsBtn.className = "tab-btn px-4 py-3 font-medium border-b-2 border-transparent text-slate-600 hover:text-slate-900 cursor-pointer";
  tabDescText.innerText = "Để xem báo cáo về quảng cáo của nhà xuất bản (có thông tin về mức độ tương tác và doanh thu ước tính của mỗi quảng cáo), hãy liên kết với một sản phẩm quảng cáo của Google.";
  
  productItems.forEach(item => {
    item.classList.toggle("hidden", item.getAttribute("data-type") !== "pub");
  });
  selectProduct("AdSense");
});

// 2. LOGIC ĐÓNG/MỞ DROPDOWN VÀ CHỌN SẢN PHẨM
const adsDropdownBtn = document.querySelector("#adsDropdownBtn");
const adsDropdownMenu = document.querySelector("#adsDropdownMenu");
const dropdownArrow = document.querySelector("#dropdownArrow");

adsDropdownBtn?.addEventListener("click", (e) => {
  e.stopPropagation();
  const isHidden = adsDropdownMenu.classList.toggle("hidden");
  dropdownArrow.innerText = isHidden ? "▼" : "▲";
});

function selectProduct(productName) {
  productItems.forEach(item => {
    const isMatch = item.getAttribute("data-name") === productName;
    item.classList.toggle("bg-blue-50/50", isMatch);
    item.querySelector(".check-icon")?.classList.toggle("hidden", !isMatch);
    if (isMatch) {
      document.querySelector(".text-product-name").innerText = productName;
    }
  });
  adsDropdownMenu.classList.add("hidden");
  dropdownArrow.innerText = "▼";
}

productItems.forEach(item => {
  item.addEventListener("click", function() {
    selectProduct(this.getAttribute("data-name"));
  });
});

document.addEventListener("click", (e) => {
  if (!e.target.closest("#adsDropdownContainer")) {
    adsDropdownMenu?.classList.add("hidden");
    if (dropdownArrow) dropdownArrow.innerText = "▼";
  }
});
