Tạo thông tin chi tiết tuỳ chỉnh

Thông tin chi tiết tuỳ chỉnh cho phép bạn tự động theo dõi hiệu quả hoạt động của thuộc tính. Tất cả người dùng trong thuộc tính này đều có thể xem thông tin chi tiết trong trang tổng quan Thông tin chi tiết khi bạn kích hoạt các điều kiện. Người dùng cũng có thể đăng ký nhận những thông tin chi tiết này qua email.

function updateSelectedButton() {
  const checkedCount = Array.from(checkboxes).filter(cb => cb.checked).length;
  btnCreateFromSelected.disabled = checkedCount === 0;
  btnCreateFromSelected.className = checkedCount > 0
    ? 'bg-[#1a73e8] hover:bg-[#1557b0] active:scale-95 text-white text-xs font-medium px-5 py-2.5 rounded transition-all cursor-pointer shadow-2xs'
    : 'bg-[#e8eaed] text-[#9aa0a6] text-xs font-medium px-5 py-2.5 rounded cursor-not-allowed transition-all';
}
checkboxes.forEach(cb => cb.addEventListener('change', updateSelectedButton));

// Thêm / Xóa Chip người nhận email
const recipientChips = document.getElementById('recipientChips');
const inputAddRecipient = document.getElementById('inputAddRecipient');
const recipientCount = document.getElementById('recipientCount');

function updateRecipientCount() {
  recipientCount.innerText = recipientChips.querySelectorAll('.recipient-chip').length;
}

recipientChips.addEventListener('click', (e) => {
  if (e.target.classList.contains('btn-remove-chip')) {
    e.target.closest('.recipient-chip').remove();
    updateRecipientCount();
  }
});

inputAddRecipient.addEventListener('keydown', (e) => {
  if (e.key === 'Enter' && inputAddRecipient.value.trim()) {
    e.preventDefault();
    const chip = document.createElement('span');
    chip.className = 'recipient-chip inline-flex items-center gap-1.5 bg-white border border-slate-300 rounded-full px-2.5 py-0.5 text-xs text-slate-700';
    chip.innerHTML = `<span>${inputAddRecipient.value.trim()}</span><button type="button" class="btn-remove-chip text-slate-400 hover:text-slate-700 cursor-pointer ml-0.5">✕</button>`;
    recipientChips.appendChild(chip);
    inputAddRecipient.value = '';
    updateRecipientCount();
  }
});

// Lưu kết quả và hiển thị ra màn hình chính
function renderCreatedItem(title, metric) {
  document.getElementById('emptyContainer').classList.add('hidden');
  document.getElementById('createdInsightsContainer').classList.remove('hidden');
  const row = document.createElement('div');
  row.className = 'px-5 py-3.5 flex items-center justify-between text-xs';
  row.innerHTML = `<div><p class="font-semibold text-slate-800">${title || 'Thông tin chi tiết mới'}</p><p class="text-slate-500 mt-0.5">Số liệu: ${metric} • Hằng ngày</p></div><span class="bg-emerald-50 text-emerald-700 px-2.5 py-0.5 rounded-full font-medium">Đang bật</span>`;
  document.getElementById('createdInsightsList').prepend(row);
}

btnCreateFromSelected.addEventListener('click', () => {
  if (btnCreateFromSelected.disabled) return;
  checkboxes.forEach(cb => {
    if (cb.checked) renderCreatedItem(cb.getAttribute('data-title'), cb.getAttribute('data-metric'));
  });
  stepDrawerOverlay.classList.add('hidden');
});

document.getElementById('btnSaveCustomInsight').addEventListener('click', () => {
  renderCreatedItem(inputInsightName.value.trim(), selectMetric.value);
  stepCreateForm.classList.add('hidden');
  stepDrawerOverlay.classList.add('hidden');
});
