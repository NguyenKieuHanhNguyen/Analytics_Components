// Toggle bật / tắt chức năng đặt lại khi có hoạt động mới
const toggleBtn = document.querySelector("#toggleResetOnActivity");
const toggleKnob = document.querySelector("#toggleKnob");
let isToggled = true;

toggleBtn.addEventListener("click", () => {
  isToggled = !isToggled;
  if (isToggled) {
    toggleBtn.classList.replace("bg-slate-300", "bg-blue-600");
    toggleKnob.classList.add("translate-x-5");
    toggleKnob.classList.remove("translate-x-0.5");
    toggleKnob.innerHTML = `<svg class="w-3.5 h-3.5 text-blue-600" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="3" d="M5 13l4 4L19 7"></path></svg>`;
  } else {
    toggleBtn.classList.replace("bg-blue-600", "bg-slate-300");
    toggleKnob.classList.remove("translate-x-5");
    toggleKnob.classList.add("translate-x-0.5");
    toggleKnob.innerHTML = '';
  }
});

// Nút lưu & huỷ
const btnSave = document.querySelector("#btnSave");
const toastSave = document.querySelector("#toastSave");
btnSave.addEventListener("click", () => {
  toastSave.classList.remove("hidden");
  setTimeout(() => toastSave.classList.add("hidden"), 2500);
});
