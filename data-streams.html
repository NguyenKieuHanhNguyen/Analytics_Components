NỘI DUNG JAVASCRIPT (DOM) 
// 1. Khai báo các phần tử DOM
const streamRows = document.querySelectorAll(".stream-row");
const streamDrawerBackdrop = document.querySelector("#streamDrawerBackdrop");
const streamDrawer = document.querySelector("#streamDrawer");
const btnCloseDrawer = document.querySelector("#btnCloseDrawer");

const detailAppName = document.querySelector("#detailAppName");
const detailPlatform = document.querySelector("#detailPlatform");
const detailPackage = document.querySelector("#detailPackage");
const detailStreamId = document.querySelector("#detailStreamId");
const detailFirebaseId = document.querySelector("#detailFirebaseId");

// 2. Mở Drawer khi click vào từng dòng luồng dữ liệu
streamRows.forEach(row => {
  row.addEventListener("click", function () {
    detailAppName.innerText = this.getAttribute("data-name");
    detailPlatform.innerText = this.getAttribute("data-platform");
    detailPackage.innerText = this.getAttribute("data-package");
    detailStreamId.innerText = this.getAttribute("data-id");
    detailFirebaseId.innerText = this.getAttribute("data-firebase");

    streamDrawerBackdrop.classList.remove("hidden");
    setTimeout(() => {
      streamDrawer.classList.remove("translate-x-full");
    }, 10);
  });
});

// 3. Hàm đóng Drawer trượt
function closeDrawer() {
  streamDrawer.classList.add("translate-x-full");
  setTimeout(() => {
    streamDrawerBackdrop.classList.add("hidden");
  }, 300);
}

btnCloseDrawer.addEventListener("click", closeDrawer);
streamDrawerBackdrop.addEventListener("click", (e) => {
  if (e.target === streamDrawerBackdrop) closeDrawer();
});

document.addEventListener("keydown", (e) => {
  if (e.key === "Escape" && !streamDrawerBackdrop.classList.contains("hidden")) {
    closeDrawer();
  }
});

// 4. Dropdown "Thêm luồng" xổ xuống
const btnAddStream = document.querySelector("#btnAddStream");
const addStreamDropdownMenu = document.querySelector("#addStreamDropdownMenu");
const arrowAddStream = document.querySelector("#arrowAddStream");
const addStreamDropdownContainer = document.querySelector("#addStreamDropdownContainer");

btnAddStream.addEventListener("click", (e) => {
  e.stopPropagation();
  const isHidden = addStreamDropdownMenu.classList.toggle("hidden");
  if (arrowAddStream) arrowAddStream.classList.toggle("rotate-180", !isHidden);
});

document.querySelectorAll(".add-stream-option").forEach(option => {
  option.addEventListener("click", function (e) {
    e.stopPropagation();
    addStreamDropdownMenu.classList.add("hidden");
    if (arrowAddStream) arrowAddStream.classList.remove("rotate-180");
  });
});

document.addEventListener("click", (e) => {
  if (addStreamDropdownContainer && !addStreamDropdownContainer.contains(e.target)) {
    addStreamDropdownMenu.classList.add("hidden");
    if (arrowAddStream) arrowAddStream.classList.remove("rotate-180");
  }
});

// 5. Tabs lọc nền tảng (Tất cả, iOS, Android, Web)
const streamTabs = document.querySelectorAll(".stream-tab-btn");
streamTabs.forEach(tab => {
  tab.addEventListener("click", function () {
    streamTabs.forEach(t => {
      t.classList.remove("text-blue-600", "border-b-2", "border-blue-600", "font-semibold");
      t.classList.add("text-slate-600");
    });
    this.classList.add("text-blue-600", "border-b-2", "border-blue-600", "font-semibold");
    this.classList.remove("text-slate-600");

    const filter = this.getAttribute("data-filter");
    streamRows.forEach(row => {
      if (filter === "all" || row.getAttribute("data-type") === filter) {
        row.classList.remove("hidden");
      } else {
        row.classList.add("hidden");
      }
    });
  });
});

// 6. Tìm kiếm luồng
const streamSearchInput = document.querySelector("#streamSearchInput");
streamSearchInput.addEventListener("input", function () {
  const query = this.value.toLowerCase().trim();
  streamRows.forEach(row => {
    const text = row.innerText.toLowerCase();
    row.classList.toggle("hidden", !text.includes(query));
  });
});
