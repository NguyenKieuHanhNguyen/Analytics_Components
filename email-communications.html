Thông tin liên lạc qua email từ Google Analytics

Thỉnh thoảng, chúng tôi gửi email để cho bạn biết thông tin mới về Google Analytics. Tuy nhiên, bạn luôn có quyền lựa chọn loại thông tin liên lạc mà bạn muốn nhận từ Google. Vì vậy, vui lòng cho chúng tôi biết lựa chọn ưu tiên của bạn bên dưới.

Bất kể lựa chọn của bạn là gì, chúng tôi vẫn phải gửi cho bạn những thông tin cập nhật quan trọng về sản phẩm ảnh hưởng đến tài khoản của bạn, nhưng đó là tất cả thông tin mà bạn sẽ nhận được từ Google. Chúng tôi tôn trọng quyền riêng tư của bạn và sẽ không chia sẻ thông tin cá nhân của bạn với các bên thứ ba hoặc đối tác.

NỘI DUNG JAVASCRIPT (DOM) 
const btnSaveEmailPref = document.querySelector("#btnSaveEmailPref");
// 1. Khai báo các phần tử DOM
const btnSaveEmailPref = document.querySelector("#btnSaveEmailPref");
const btnCancelEmailPref = document.querySelector("#btnCancelEmailPref");
const emailCheckboxes = document.querySelectorAll('input[name="email_pref"]');

// 2. Lưu lại trạng thái ban đầu của các checkbox
const initialStates = Array.from(emailCheckboxes).map(cb => cb.checked);

// 3. Hàm kiểm tra xem người dùng có thay đổi lựa chọn hay không
function checkDirtyState() {
  const isChanged = Array.from(emailCheckboxes).some((cb, index) => cb.checked !== initialStates[index]);
  
  btnSaveEmailPref.disabled = !isChanged;
  btnCancelEmailPref.disabled = !isChanged;

  if (isChanged) {
    btnSaveEmailPref.classList.remove("disabled:bg-slate-200", "disabled:text-slate-400");
  }
}

// Lắng nghe sự kiện change trên từng checkbox
emailCheckboxes.forEach(cb => {
  cb.addEventListener("change", checkDirtyState);
});

// 4. Xử lý nút "Lưu thay đổi"
btnSaveEmailPref.addEventListener("click", () => {
  if (btnSaveEmailPref.disabled) return;

  const currentValues = Array.from(emailCheckboxes).map(cb => cb.checked);
  console.log("Đã cập nhật cài đặt email:", currentValues);

  // Cập nhật lại mốc trạng thái ban đầu sau khi lưu
  currentValues.forEach((val, idx) => { initialStates[idx] = val; });
  
  // Hiệu ứng thông báo lưu thành công
  const originText = btnSaveEmailPref.innerText;
  btnSaveEmailPref.innerText = "Đã lưu thành công!";
  btnSaveEmailPref.classList.replace("bg-blue-600", "bg-emerald-600");

  setTimeout(() => {
    btnSaveEmailPref.innerText = originText;
    btnSaveEmailPref.classList.replace("bg-emerald-600", "bg-blue-600");
    checkDirtyState();
  }, 1500);
});

// 5. Xử lý nút "Huỷ" -> Phục hồi lại trạng thái trước khi chỉnh sửa
btnCancelEmailPref.addEventListener("click", () => {
  if (btnCancelEmailPref.disabled) return;

  emailCheckboxes.forEach((cb, index) => {
    cb.checked = initialStates[index];
  });
  checkDirtyState();
});
