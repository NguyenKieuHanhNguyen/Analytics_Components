// Mở/Đóng accordion khi click vào từng hàng sự kiện
document.querySelectorAll('.event-header').forEach(header => {
  header.addEventListener('click', () => {
    const content = header.nextElementSibling;
    if (content) {
      content.classList.toggle('hidden');
    }
  });
});

// Nút đóng chi tiết trong khung
document.querySelectorAll('.btn-close-detail').forEach(btn => {
  btn.addEventListener('click', (e) => {
    const content = e.target.closest('.event-content');
    if (content) content.classList.add('hidden');
  });
});

// Toggle dropdown cấu hình tuỳ chỉnh
const btnToggleSplitMenu = document.querySelector('#btnToggleSplitMenu');
const splitDropdownMenu = document.querySelector('#splitDropdownMenu');
btnToggleSplitMenu?.addEventListener('click', () => {
  splitDropdownMenu?.classList.toggle('hidden');
});

// Chuyển đổi mã theo Tab ngôn ngữ
const snippetMap = {
  java: `Bundle bundle = new Bundle();\nbundle.putString(FirebaseAnalytics.Param.METHOD, method);\nmFirebaseAnalytics.logEvent(FirebaseAnalytics.Event.SIGN_UP, bundle);`,
  kotlin: `val bundle = Bundle()\nbundle.putString(FirebaseAnalytics.Param.METHOD, method)\nfirebaseAnalytics.logEvent(FirebaseAnalytics.Event.SIGN_UP, bundle)`,
  objc: `[FIRAnalytics logEventWithName:kFIREventSignUp parameters:@{\n  kFIRParameterMethod : method\n}];`,
  swift: `Analytics.logEvent(AnalyticsEventSignUp, parameters: [\n  AnalyticsParameterMethod: method\n])`,
  unity: `FirebaseAnalytics.LogEvent(\n  FirebaseAnalytics.EventSignUp,\n  new Parameter(FirebaseAnalytics.ParameterMethod, method)\n);`,
  cpp: `firebase::analytics::Parameter params[] = {\n  firebase::analytics::Parameter(firebase::analytics::kParameterMethod, method)\n};\nfirebase::analytics::LogEvent(firebase::analytics::kEventSignUp, params, 1);`
};

document.querySelectorAll('.code-tab').forEach(tab => {
  tab.addEventListener('click', () => {
    document.querySelectorAll('.code-tab').forEach(t => {
      t.classList.remove('border-blue-600', 'text-blue-600', 'font-semibold');
      t.classList.add('border-transparent', 'text-slate-600');
    });
    tab.classList.add('border-blue-600', 'text-blue-600', 'font-semibold');
    tab.classList.remove('border-transparent', 'text-slate-600');

    const lang = tab.getAttribute('data-lang');
    const snippetBox = document.querySelector('#codeSnippetBox');
    if (snippetBox && snippetMap[lang]) {
      snippetBox.innerText = snippetMap[lang];
    }
  });
});

// Nút sao chép mã sự kiện
document.querySelectorAll('.btn-copy-code').forEach(btn => {
  btn.addEventListener('click', (e) => {
    const box = e.target.closest('.event-content')?.querySelector('div[class*="font-mono"]');
    if (box) {
      navigator.clipboard.writeText(box.innerText).then(() => {
        const originText = btn.innerText;
        btn.innerText = 'Đã sao chép!';
        setTimeout(() => { btn.innerText = originText; }, 1500);
      });
    }
  });
});
