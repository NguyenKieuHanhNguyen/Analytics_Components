NỘI DUNG JAVASCRIPT (DOM) 
// 1. Khai báo container carousel và 2 nút bấm
const carouselSlider = document.querySelector("#carouselSlider");
const btnPrev = document.querySelector("#btnPrev");
const btnNext = document.querySelector("#btnNext");

// Khoảng cách cuộn mỗi lần bấm (khoảng 1-2 thẻ: 260px)
const SCROLL_AMOUNT = 260;

// 2. Hàm kiểm tra vị trí cuộn để ẩn/hiện nút điều hướng
function updateButtonVisibility() {
  const scrollLeft = carouselSlider.scrollLeft;
  const maxScrollLeft = carouselSlider.scrollWidth - carouselSlider.clientWidth;

  // Nếu đã ở đầu danh sách -> Ẩn nút Trái (<)
  if (scrollLeft <= 10) {
    btnPrev.classList.add("hidden");
  } else {
    btnPrev.classList.remove("hidden");
  }

  // Nếu đã ở cuối danh sách -> Ẩn nút Phải (>)
  if (scrollLeft >= maxScrollLeft - 10) {
    btnNext.classList.add("hidden");
  } else {
    btnNext.classList.remove("hidden");
  }
}

// 3. Sự kiện bấm nút Next & Prev
btnNext.addEventListener("click", () => {
  carouselSlider.scrollBy({ left: SCROLL_AMOUNT, behavior: "smooth" });
});

btnPrev.addEventListener("click", () => {
  carouselSlider.scrollBy({ left: -SCROLL_AMOUNT, behavior: "smooth" });
});

// 4. Lắng nghe sự kiện cuộn để cập nhật trạng thái nút liên tục
carouselSlider.addEventListener("scroll", updateButtonVisibility);

// Khởi tạo trạng thái ban đầu
updateButtonVisibility();

// 1. KHAI BÁO CÁC PHẦN TỬ DOM
const collectionTrack = document.querySelector("#collectionTrack");
const btnPrevCollection = document.querySelector("#btnPrevCollection");
const btnNextCollection = document.querySelector("#btnNextCollection");
const btnCreateCollection = document.querySelector("#btnCreateCollection");

// 2. LOGIC CUỘN MƯỢT VÀ ẨN/HIỆN NÚT ĐIỀU HƯỚNG TRÁI / PHẢI
const scrollStep = 280; // Độ rộng 1 card + khoảng cách gap

function updateScrollButtons() {
  if (!collectionTrack || !btnPrevCollection || !btnNextCollection) return;
  
  const currentScroll = collectionTrack.scrollLeft;
  const maxScroll = collectionTrack.scrollWidth - collectionTrack.clientWidth;
  
  btnPrevCollection.classList.toggle("hidden", currentScroll <= 5);
  btnNextCollection.classList.toggle("hidden", currentScroll >= maxScroll - 5);
}

if (collectionTrack && btnPrevCollection && btnNextCollection) {
  btnNextCollection.addEventListener("click", () => {
    collectionTrack.scrollBy({ left: scrollStep, behavior: "smooth" });
  });

  btnPrevCollection.addEventListener("click", () => {
    collectionTrack.scrollBy({ left: -scrollStep, behavior: "smooth" });
  });

  collectionTrack.addEventListener("scroll", updateScrollButtons);
  window.addEventListener("resize", updateScrollButtons);
  updateScrollButtons();
}

// 3. TÍNH NĂNG KÉO RÊ CHUỘT ĐỂ CUỘN (DRAG-TO-SCROLL CHO DESKTOP)
let isDragging = false;
let startX = 0;
let scrollStart = 0;

if (collectionTrack) {
  collectionTrack.addEventListener("mousedown", (e) => {
    if (e.target.closest("button") || e.target.closest(".collection-dropdown-container")) return;
    
    isDragging = true;
    collectionTrack.classList.add("cursor-grabbing");
    startX = e.pageX - collectionTrack.offsetLeft;
    scrollStart = collectionTrack.scrollLeft;
  });

  document.addEventListener("mousemove", (e) => {
    if (!isDragging) return;
    e.preventDefault();
    const x = e.pageX - collectionTrack.offsetLeft;
    const walk = (x - startX) * 1.5;
    collectionTrack.scrollLeft = scrollStart - walk;
  });

  document.addEventListener("mouseup", () => {
    if (!isDragging) return;
    isDragging = false;
    collectionTrack.classList.remove("cursor-grabbing");
  });
}

// 4. XỬ LÝ MENU TÙY CHỌN 3 CHẤM (⋮) TRÊN TỪNG THẺ
const collectionMoreBtns = document.querySelectorAll(".btn-collection-more");

collectionMoreBtns.forEach(btn => {
  btn.addEventListener("click", function (e) {
    e.stopPropagation();
    const currentMenu = this.nextElementSibling;
    
    document.querySelectorAll(".collection-dropdown-menu").forEach(menu => {
      if (menu !== currentMenu) menu.classList.add("hidden");
    });
    
    if (currentMenu) {
      currentMenu.classList.toggle("hidden");
    }
  });
});

// 5. ĐÓNG MENU KHI CLICK RA NGOÀI HOẶC BẤM ESCAPE
document.addEventListener("click", (e) => {
  if (!e.target.closest(".collection-dropdown-container")) {
    document.querySelectorAll(".collection-dropdown-menu").forEach(menu => {
      menu.classList.add("hidden");
    });
  }
});

document.addEventListener("keydown", (e) => {
  if (e.key === "Escape") {
    document.querySelectorAll(".collection-dropdown-menu").forEach(menu => {
      menu.classList.add("hidden");
    });
  }
});
