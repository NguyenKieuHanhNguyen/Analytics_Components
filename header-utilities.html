NỘI DUNG JAVASCRIPT (DOM) 
// 1. Đóng/Mở từng Popover và tự ẩn Tooltip khi Popover đang mở
const triggers = document.querySelectorAll('[data-popover-trigger]');
const panels = document.querySelectorAll('.popover-panel');

triggers.forEach(btn => {
  btn.addEventListener('click', (e) => {
    e.stopPropagation();
    const targetId = btn.getAttribute('data-popover-trigger');
    const targetPanel = document.getElementById(targetId);
    const isHidden = targetPanel.classList.contains('hidden');

    // Đóng tất cả popover khác
    panels.forEach(p => p.classList.add('hidden'));
    document.querySelectorAll('.tooltip-box').forEach(t => t.classList.remove('!hidden'));

    if (isHidden) {
      targetPanel.classList.remove('hidden');
      const tooltip = btn.parentElement.querySelector('.tooltip-box');
      if (tooltip) tooltip.classList.add('!hidden');
    }
  });
});

// Ngăn click bên trong Popover làm đóng menu
panels.forEach(panel => {
  panel.addEventListener('click', (e) => e.stopPropagation());
});

// Click ra ngoài để đóng toàn bộ Popover
document.addEventListener('click', () => {
  panels.forEach(p => p.classList.add('hidden'));
  document.querySelectorAll('.tooltip-box').forEach(t => t.classList.remove('!hidden'));
});

// 2. Accordion "Khám phá thêm các sản phẩm khác"
const btnMoreProducts = document.getElementById('btnMoreProducts');
const moreProductsList = document.getElementById('moreProductsList');
const arrowMoreProducts = document.getElementById('arrowMoreProducts');

btnMoreProducts?.addEventListener('click', () => {
  moreProductsList.classList.toggle('hidden');
  arrowMoreProducts.classList.toggle('rotate-180');
});
