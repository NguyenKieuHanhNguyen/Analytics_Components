const btnOpenAnnotations = document.getElementById('btnOpenAnnotations');
const btnOpenInsights = document.getElementById('btnOpenInsights');
const drawerAnnotations = document.getElementById('drawerAnnotations');
const drawerInsights = document.getElementById('drawerInsights');

function setActiveTab(activeBtn, inactiveBtn, activeDrawer, inactiveDrawer) {
  inactiveDrawer.classList.add('hidden');
  activeDrawer.classList.toggle('hidden');

  inactiveBtn.classList.remove('bg-[#d2e3fc]', 'text-[#1967d2]');
  inactiveBtn.classList.add('text-[#5f6368]');

  if (!activeDrawer.classList.contains('hidden')) {
    activeBtn.classList.add('bg-[#d2e3fc]', 'text-[#1967d2]');
    activeBtn.classList.remove('text-[#5f6368]');
  } else {
    activeBtn.classList.remove('bg-[#d2e3fc]', 'text-[#1967d2]');
    activeBtn.classList.add('text-[#5f6368]');
  }
}

btnOpenAnnotations.addEventListener('click', () => {
  setActiveTab(btnOpenAnnotations, btnOpenInsights, drawerAnnotations, drawerInsights);
});
btnOpenInsights.addEventListener('click', () => {
  setActiveTab(btnOpenInsights, btnOpenAnnotations, drawerInsights, drawerAnnotations);
});

document.querySelectorAll('.btn-close-drawer').forEach(btn => {
  btn.addEventListener('click', () => {
    drawerAnnotations.classList.add('hidden');
    drawerInsights.classList.add('hidden');
    [btnOpenAnnotations, btnOpenInsights].forEach(b => {
      b.classList.remove('bg-[#d2e3fc]', 'text-[#1967d2]');
      b.classList.add('text-[#5f6368]');
    });
  });
});

const annTitleInput = document.getElementById('annTitleInput');
const annDescInput = document.getElementById('annDescInput');
const annDateInput = document.getElementById('annDateInput');
const btnSubmitAnnotation = document.getElementById('btnSubmitAnnotation');
let selectedAnnColor = '#7b1fa2';

function validateAnnotationForm() {
  const isValid = annTitleInput.value.trim() !== '' && annDateInput.value.trim() !== '';
  btnSubmitAnnotation.disabled = !isValid;
  btnSubmitAnnotation.className = isValid
    ? 'px-5 py-2 rounded text-xs font-medium bg-[#1a73e8] hover:bg-[#1557b0] text-white cursor-pointer transition-all shadow-2xs'
    : 'px-5 py-2 rounded text-xs font-medium bg-[#e8eaed] text-[#9aa0a6] cursor-not-allowed transition-all';
}

annTitleInput.addEventListener('input', () => {
  document.getElementById('annTitleCount').innerText = annTitleInput.value.length;
  validateAnnotationForm();
});
annDescInput.addEventListener('input', () => {
  document.getElementById('annDescCount').innerText = annDescInput.value.length;
});
annDateInput.addEventListener('input', validateAnnotationForm);

document.getElementById('btnPickTodayDate').addEventListener('click', () => {
  annDateInput.value = new Date().toLocaleDateString('vi-VN');
  validateAnnotationForm();
});

document.querySelectorAll('.ann-color-btn').forEach(btn => {
  btn.addEventListener('click', () => {
    document.querySelectorAll('.check-mark').forEach(c => c.classList.add('hidden'));
    btn.querySelector('.check-mark').classList.remove('hidden');
    selectedAnnColor = btn.getAttribute('data-color');
  });
});

btnSubmitAnnotation.addEventListener('click', () => {
  if (btnSubmitAnnotation.disabled) return;
  document.getElementById('emptyAnnotationText').classList.add('hidden');
  const card = document.createElement('div');
  card.className = 'p-3 rounded-lg border border-slate-200 bg-slate-50/80 text-xs space-y-1.5 border-l-4';
  card.style.borderLeftColor = selectedAnnColor;
  card.innerHTML = `<div class="flex justify-between font-semibold text-slate-800"><span>${annTitleInput.value.trim()}</span><span class="text-[11px] font-normal text-slate-400">${annDateInput.value.trim()}</span></div>${annDescInput.value.trim() ? `<p class="text-slate-600">${annDescInput.value.trim()}</p>` : ''}`;
  document.getElementById('annotationList').prepend(card);

  annTitleInput.value = ''; annDescInput.value = ''; annDateInput.value = '';
  document.getElementById('annTitleCount').innerText = '0';
  document.getElementById('annDescCount').innerText = '0';
  validateAnnotationForm();
  viewAnnotationCreate.classList.add('hidden');
  viewAnnotationList.classList.remove('hidden');
});

  sw.addEventListener('click', () => {

    const knob = sw.querySelector('.switch-knob');
    if (!isChecked) {
      sw.setAttribute('data-checked', 'true');
      sw.classList.replace('bg-slate-300', 'bg-[#8ab4f8]');
      knob.classList.replace('bg-[#3c4043]', 'bg-[#1a73e8]');
      knob.classList.replace('bg-[#5f6368]', 'bg-[#1a73e8]');
      knob.classList.add('translate-x-4');
      knob.innerText = '✓';
    } else {
      sw.setAttribute('data-checked', 'false');
      sw.classList.replace('bg-[#8ab4f8]', 'bg-slate-300');
      knob.classList.replace('bg-[#1a73e8]', 'bg-[#5f6368]');
      knob.classList.remove('translate-x-4');
      knob.innerText = '−';
    }
  });
});
