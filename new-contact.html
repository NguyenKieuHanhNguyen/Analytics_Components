NỘI DUNG JAVASCRIPT (DOM) 
const fieldBoxes = document.querySelectorAll('.field-box');
const btnSubmit = document.querySelector('#btnSubmitContact');

function setFieldState(box, state) {
  const input = box.querySelector('.contact-input');
  const label = box.querySelector('.field-label');

  input.classList.remove(
    'border-[#d93025]', 'border-[#1a73e8]', 'border-slate-400',
    'border-b', 'border-b-2',
    'bg-[#e8eaed]', 'bg-[#f1f3f4]'
  );
  label.classList.remove('text-[#d93025]', 'text-[#1a73e8]', 'text-slate-600');

  if (state === 'active') {
    input.classList.add('border-b-2', 'border-[#1a73e8]', 'bg-[#e8eaed]');
    label.classList.add('text-[#1a73e8]');
  } else if (state === 'error') {
    input.classList.add('border-b-2', 'border-[#d93025]', 'bg-[#f1f3f4]');
    label.classList.add('text-[#d93025]');
  } else {
    input.classList.add('border-b', 'border-slate-400', 'bg-[#f1f3f4]');
    label.classList.add('text-slate-600');
  }
}

fieldBoxes.forEach(box => {
  const input = box.querySelector('.contact-input');
  let isTouched = false;

  box.addEventListener('mouseenter', () => setFieldState(box, 'active'));
  box.addEventListener('mouseleave', () => {
    if (document.activeElement !== input) {
      setFieldState(box, input.value.trim() ? 'normal' : (isTouched ? 'error' : 'normal'));
    }
  });
  input.addEventListener('focus', () => {
    isTouched = true;
    setFieldState(box, 'active');
  });
  input.addEventListener('blur', () => {
    setFieldState(box, input.value.trim() ? 'normal' : 'error');
  });
  input.addEventListener('input', checkFormValid);
});

function checkFormValid() {
  const inputs = document.querySelectorAll('.contact-input');
  const allFilled = Array.from(inputs).every(i => i.value.trim() !== '');
  btnSubmit.disabled = !allFilled;
  btnSubmit.className = allFilled
    ? 'px-5 py-2 rounded text-sm font-medium bg-[#1a73e8] hover:bg-[#1557b0] text-white cursor-pointer transition-all shadow-xs'
    : 'px-5 py-2 rounded text-sm font-medium bg-[#e8eaed] text-[#9aa0a6] cursor-not-allowed transition-all';
}
