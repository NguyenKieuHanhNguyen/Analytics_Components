const btnOpenAccountModal = document.querySelector("#btnOpenAccountModal");
const gaModalBackdrop = document.querySelector("#gaModalBackdrop");
const btnCloseModal = document.querySelector("#btnCloseModal");
const currentAccountText = document.querySelector("#currentAccountText");
const currentPropertyText = document.querySelector("#currentPropertyText");

const btnAccountDropdown = document.querySelector("#btnAccountDropdown");
const accountDropdownMenu = document.querySelector("#accountDropdownMenu");
const dropdownArrow = document.querySelector("#dropdownArrow");
const selectedAccountLabel = document.querySelector("#selectedAccountLabel");
const accountDropdownContainer = document.querySelector("#accountDropdownContainer");

// 1. Mở / Đóng Modal
btnOpenAccountModal.addEventListener("click", () => gaModalBackdrop.classList.remove("hidden"));
btnCloseModal.addEventListener("click", () => gaModalBackdrop.classList.add("hidden"));

// Đóng khi click ra ngoài backdrop
gaModalBackdrop.addEventListener("click", (e) => {
  if (e.target === gaModalBackdrop) gaModalBackdrop.classList.add("hidden");
});

// Đóng bằng phím Escape (ESC) trên bàn phím
document.addEventListener("keydown", (e) => {
  if (e.key === "Escape" && !gaModalBackdrop.classList.contains("hidden")) {
    gaModalBackdrop.classList.add("hidden");
  }
});

// 2. Dropdown "Tất cả tài khoản" xổ xuống
btnAccountDropdown.addEventListener("click", (e) => {
  e.stopPropagation();
  const isHidden = accountDropdownMenu.classList.toggle("hidden");
  dropdownArrow.classList.toggle("rotate-180", !isHidden);
});

// Chọn tài khoản từ danh sách xổ xuống
document.querySelectorAll(".dropdown-account-item").forEach(item => {
  item.addEventListener("click", function (e) {
    e.stopPropagation();
    selectedAccountLabel.innerText = this.getAttribute("data-account-name");
    document.querySelectorAll(".dropdown-account-item .item-check").forEach(c => c.classList.add("hidden"));
    this.querySelector(".item-check").classList.remove("hidden");
    accountDropdownMenu.classList.add("hidden");
    dropdownArrow.classList.remove("rotate-180");
  });
});

// Click ra ngoài bất kỳ đâu thì tự đóng Dropdown tài khoản
document.addEventListener("click", (e) => {
  if (accountDropdownContainer && !accountDropdownContainer.contains(e.target)) {
    accountDropdownMenu.classList.add("hidden");
    dropdownArrow.classList.remove("rotate-180");
  }
});

// 3. Chọn thuộc tính con -> Cập nhật Header và đóng Popup
document.querySelectorAll(".property-select-item").forEach(item => {
  item.addEventListener("click", function () {
    currentPropertyText.innerText = this.getAttribute("data-property");
    currentAccountText.innerText = this.getAttribute("data-account");
    gaModalBackdrop.classList.add("hidden");
  });
});

// 4. Bật / Tắt đánh dấu sao yêu thích ⭐
document.querySelectorAll(".btn-star").forEach(star => {
  star.addEventListener("click", function (e) {
    e.stopPropagation(); // Không kích hoạt chọn dòng
    this.classList.toggle("text-amber-400");
    this.classList.toggle("text-slate-300");
  });
});

const metricPickerDropdown = document.querySelector("#metricPickerDropdown");
const metricCardArrow = document.querySelector("#metricCardArrow");
const activeMetricTitle = document.querySelector("#activeMetricTitle");
const metricCardWrapper = document.querySelector("#metricCardWrapper");

// 1. Mở / Đóng Dropdown chọn chỉ số
btnOpenMetricPicker.addEventListener("click", (e) => {
  e.stopPropagation();
  const isHidden = metricPickerDropdown.classList.toggle("hidden");
  metricCardArrow.classList.toggle("rotate-180", !isHidden);
});

// 2. Đổi danh mục cột trái
const metricCategoryBtns = document.querySelectorAll(".metric-category-btn");
metricCategoryBtns.forEach(btn => {
  btn.addEventListener("click", function (e) {
    e.stopPropagation();
    metricCategoryBtns.forEach(b => {
      b.classList.remove("bg-blue-50/70", "text-blue-700", "font-semibold");
      b.classList.add("text-slate-600");
    });
    this.classList.add("bg-blue-50/70", "text-blue-700", "font-semibold");
    this.classList.remove("text-slate-600");
  });
});

// 3. Chọn chỉ số cột phải -> Cập nhật lên Card và đóng Dropdown
document.querySelectorAll(".metric-item-btn[data-metric-name]").forEach(item => {
  item.addEventListener("click", function (e) {
    e.stopPropagation();
    const metricName = this.getAttribute("data-metric-name");
    activeMetricTitle.innerText = metricName;
    metricPickerDropdown.classList.add("hidden");
    metricCardArrow.classList.remove("rotate-180");
  });
});

// 4. Click ra ngoài vùng Card thì tự động đóng Popup
document.addEventListener("click", (e) => {
  if (metricCardWrapper && !metricCardWrapper.contains(e.target)) {
    metricPickerDropdown.classList.add("hidden");
    metricCardArrow.classList.remove("rotate-180");
  }
});
