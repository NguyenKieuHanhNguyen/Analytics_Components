NỘI DUNG JAVASCRIPT (DOM) 
const btnSaveProperty = document.querySelector("#btnSaveProperty");
const btnCancelProperty = document.querySelector("#btnCancelProperty");
const formControls = document.querySelectorAll(".form-control");
const btnCopyPropertyId = document.querySelector("#btnCopyPropertyId");
const propertyIdText = document.querySelector("#propertyIdText");

// 1. Lưu giá trị khởi tạo ban đầu để so sánh thay đổi
const initialValues = Array.from(formControls).map(input => input.value);

function checkDirtyState() {
  const isChanged = Array.from(formControls).some((input, idx) => input.value !== initialValues[idx]);
  
  btnSaveProperty.disabled = !isChanged;
  if (isChanged) {
    btnSaveProperty.classList.remove("bg-slate-200", "text-slate-400", "cursor-not-allowed");
    btnSaveProperty.classList.add("bg-blue-600", "text-white", "hover:bg-blue-700", "cursor-pointer");
  } else {
    btnSaveProperty.classList.add("bg-slate-200", "text-slate-400", "cursor-not-allowed");
    btnSaveProperty.classList.remove("bg-blue-600", "text-white", "hover:bg-blue-700", "cursor-pointer");
  }
}

// Bắt sự kiện thay đổi trên các input và selectbox
formControls.forEach(ctrl => {
  ctrl.addEventListener("input", checkDirtyState);
  ctrl.addEventListener("change", checkDirtyState);
});

// 2. Nút Lưu
btnSaveProperty.addEventListener("click", () => {
  if (btnSaveProperty.disabled) return;
  console.log("Đã cập nhật cấu hình tài sản!");

  // Cập nhật lại mốc ban đầu
  formControls.forEach((input, idx) => initialValues[idx] = input.value);
  
  const originText = btnSaveProperty.innerText;
  btnSaveProperty.innerText = "Đã lưu!";
  btnSaveProperty.classList.replace("bg-blue-600", "bg-emerald-600");

  setTimeout(() => {
    btnSaveProperty.innerText = originText;
    btnSaveProperty.classList.replace("bg-emerald-600", "bg-blue-600");
    checkDirtyState();
  }, 1400);
});

// 3. Nút Huỷ: Khôi phục lại toàn bộ dữ liệu ban đầu
btnCancelProperty.addEventListener("click", () => {
  formControls.forEach((input, idx) => {
    input.value = initialValues[idx];
  });
  checkDirtyState();
});

// 4. Nút Sao chép Mã Tài Sản
btnCopyPropertyId.addEventListener("click", () => {
  navigator.clipboard.writeText(propertyIdText.innerText).then(() => {
    alert("Đã sao chép mã tài sản: " + propertyIdText.innerText);
  });
});
