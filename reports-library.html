NỘI DUNG JAVASCRIPT (DOM) 
// 1. TÌM KIẾM BÁO CÁO THEO TÊN
const reportSearchInput = document.querySelector("#reportSearchInput");
const reportRows = document.querySelectorAll(".report-row");

if (reportSearchInput) {
  reportSearchInput.addEventListener("input", function () {
    const q = this.value.toLowerCase().trim();
    reportRows.forEach(row => {
      const title = row.querySelector(".report-title")?.innerText.toLowerCase() || "";
      row.classList.toggle("hidden", !title.includes(q));
    });
  });
}

// 2. CHỌN TẤT CẢ DÒNG BÁO CÁO (SELECT ALL CHECKBOX)
const selectAllReports = document.querySelector("#selectAllReports");
const reportCheckboxes = document.querySelectorAll(".report-checkbox");

if (selectAllReports) {
  selectAllReports.addEventListener("change", function () {
    reportCheckboxes.forEach(cb => {
      cb.checked = this.checked;
      cb.closest(".report-row").classList.toggle("bg-blue-50/40", this.checked);
    });
  });
}

reportCheckboxes.forEach(cb => {
  cb.addEventListener("change", function () {
    this.closest(".report-row").classList.toggle("bg-blue-50/40", this.checked);
    if (selectAllReports) {
      const allChecked = Array.from(reportCheckboxes).every(c => c.checked);
      selectAllReports.checked = allChecked;
    }
  });
});

// 3. MENU TÙY CHỌN 3 CHẤM (⋮) TRÊN TỪNG DÒNG
document.querySelectorAll(".btn-report-more").forEach(btn => {
  btn.addEventListener("click", function (e) {
    e.stopPropagation();
    const currentMenu = this.nextElementSibling;
    document.querySelectorAll(".report-dropdown-menu").forEach(menu => {
      if (menu !== currentMenu) menu.classList.add("hidden");
    });
    if (currentMenu) currentMenu.classList.toggle("hidden");
  });
});

document.addEventListener("click", (e) => {
  if (!e.target.closest(".report-dropdown-container")) {
    document.querySelectorAll(".report-dropdown-menu").forEach(menu => {
      menu.classList.add("hidden");
    });
  }
});

// 4. LOGIC ĐIỀU KHIỂN PHÂN TRANG (PAGINATION)
const pageSizeSelect = document.querySelector("#pageSizeSelect");
const paginationInfo = document.querySelector("#paginationInfo");
const btnFirstPage = document.querySelector("#btnFirstPage");
const btnPrevPage = document.querySelector("#btnPrevPage");
const btnNextPage = document.querySelector("#btnNextPage");
const btnLastPage = document.querySelector("#btnLastPage");

let currentPage = 1;
const totalItems = 31;
let pageSize = parseInt(pageSizeSelect?.value || 15);

function updatePaginationUI() {
  if (!paginationInfo) return;
  const totalPages = Math.ceil(totalItems / pageSize);
  const start = (currentPage - 1) * pageSize + 1;
  const end = Math.min(currentPage * pageSize, totalItems);

  paginationInfo.innerText = `${start} – ${end}/${totalItems}`;

  const isFirst = currentPage === 1;
  const isLast = currentPage === totalPages;

  [btnFirstPage, btnPrevPage].forEach(btn => {
    if (!btn) return;
    btn.disabled = isFirst;
    btn.classList.toggle("text-slate-300", isFirst);
    btn.classList.toggle("cursor-not-allowed", isFirst);
    btn.classList.toggle("text-slate-600", !isFirst);
    btn.classList.toggle("hover:bg-slate-100", !isFirst);
    btn.classList.toggle("cursor-pointer", !isFirst);
  });

  [btnNextPage, btnLastPage].forEach(btn => {
    if (!btn) return;
    btn.disabled = isLast;
    btn.classList.toggle("text-slate-300", isLast);
    btn.classList.toggle("cursor-not-allowed", isLast);
    btn.classList.toggle("text-slate-600", !isLast);
    btn.classList.toggle("hover:bg-slate-100", !isLast);
    btn.classList.toggle("cursor-pointer", !isLast);
  });
}

pageSizeSelect?.addEventListener("change", function () {
  pageSize = parseInt(this.value);
  currentPage = 1;
  updatePaginationUI();
});

btnNextPage?.addEventListener("click", () => {
  const totalPages = Math.ceil(totalItems / pageSize);
  if (currentPage < totalPages) {
    currentPage++;
    updatePaginationUI();
  }
});

btnPrevPage?.addEventListener("click", () => {
  if (currentPage > 1) {
    currentPage--;
    updatePaginationUI();
  }
});

btnFirstPage?.addEventListener("click", () => {
  currentPage = 1;
  updatePaginationUI();
});

btnLastPage?.addEventListener("click", () => {
  currentPage = Math.ceil(totalItems / pageSize);
  updatePaginationUI();
});
