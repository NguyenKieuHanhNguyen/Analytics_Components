// 1. Dropdown Ngôn ngữ
const btnLangDropdown = document.querySelector("#btnLangDropdown");
const langMenu = document.querySelector("#langMenu");
const arrowLang = document.querySelector("#arrowLang");
const langDropdownContainer = document.querySelector("#langDropdownContainer");
const currentLangText = document.querySelector("#currentLangText");
const langSearchInput = document.querySelector("#langSearchInput");

btnLangDropdown.addEventListener("click", (e) => {
  e.stopPropagation();
  dateMenu.classList.add("hidden");
  arrowDate.classList.remove("rotate-180");
  
  const isHidden = langMenu.classList.toggle("hidden");
  arrowLang.classList.toggle("rotate-180", !isHidden);
  if (!isHidden) langSearchInput.focus();
});

// Lọc tìm kiếm ngôn ngữ
langSearchInput.addEventListener("input", function () {
  const q = this.value.toLowerCase().trim();
  document.querySelectorAll(".lang-option").forEach(opt => {
    opt.classList.toggle("hidden", !opt.innerText.toLowerCase().includes(q));
  });
});

// Chọn ngôn ngữ
document.querySelectorAll(".lang-option").forEach(opt => {
  opt.addEventListener("click", function (e) {
    e.stopPropagation();
    currentLangText.innerText = this.innerText.trim();
    langMenu.classList.add("hidden");
    arrowLang.classList.remove("rotate-180");
  });
});

// 2. Dropdown Phạm vi ngày mặc định
const btnDateDropdown = document.querySelector("#btnDateDropdown");
const dateMenu = document.querySelector("#dateMenu");
const arrowDate = document.querySelector("#arrowDate");
const dateDropdownContainer = document.querySelector("#dateDropdownContainer");
const currentDateText = document.querySelector("#currentDateText");

btnDateDropdown.addEventListener("click", (e) => {
  e.stopPropagation();
  langMenu.classList.add("hidden");
  arrowLang.classList.remove("rotate-180");

  const isHidden = dateMenu.classList.toggle("hidden");
  arrowDate.classList.toggle("rotate-180", !isHidden);
});

// Chọn phạm vi ngày
document.querySelectorAll(".date-option").forEach(opt => {
  opt.addEventListener("click", function (e) {
    e.stopPropagation();
    const val = this.getAttribute("data-val");
    currentDateText.innerText = val;

    // Cập nhật lại dấu tích xanh ✓ và style active
    document.querySelectorAll(".date-option").forEach(o => {
      o.classList.remove("bg-blue-50/60", "text-blue-700", "font-medium");
      o.classList.add("text-slate-700");
      o.querySelector(".check-icon").classList.add("hidden");
    });
    this.classList.add("bg-blue-50/60", "text-blue-700", "font-medium");
    this.classList.remove("text-slate-700");
    this.querySelector(".check-icon").classList.remove("hidden");

    dateMenu.classList.add("hidden");
    arrowDate.classList.remove("rotate-180");
  });
});

// 3. Đóng menu khi click ra ngoài màn hình
document.addEventListener("click", (e) => {
  if (langDropdownContainer && !langDropdownContainer.contains(e.target)) {
    langMenu.classList.add("hidden");
    arrowLang.classList.remove("rotate-180");
  }
  if (dateDropdownContainer && !dateDropdownContainer.contains(e.target)) {
    dateMenu.classList.add("hidden");
    arrowDate.classList.remove("rotate-180");
  }
});

const defaultToolbar = document.querySelector("#defaultToolbar");
const selectionToolbar = document.querySelector("#selectionToolbar");
const selectedCountText = document.querySelector("#selectedCountText");
const btnClearSelection = document.querySelector("#btnClearSelection");
const selectAllUsers = document.querySelector("#selectAllUsers");
const userCheckboxes = document.querySelectorAll(".user-checkbox");
const userRows = document.querySelectorAll(".user-row");
const userSearchInput = document.querySelector("#userSearchInput");

function updateSelectionState() {
  const checkedBoxes = Array.from(userCheckboxes).filter(cb => cb.checked);
  const total = userCheckboxes.length;
  const count = checkedBoxes.length;

  if (count > 0) {
    defaultToolbar?.classList.add("hidden");
    selectionToolbar?.classList.remove("hidden");
    if (selectedCountText) selectedCountText.innerText = `Đã chọn ${count}/${total}`;
    if (selectAllUsers) selectAllUsers.checked = (count === total);
  } else {
    defaultToolbar?.classList.remove("hidden");
    selectionToolbar?.classList.add("hidden");
    if (selectAllUsers) selectAllUsers.checked = false;
  }
}

userCheckboxes.forEach(cb => {
  cb.addEventListener("change", function() {
    this.closest(".user-row").classList.toggle("bg-blue-50/40", this.checked);
    updateSelectionState();
  });
});

selectAllUsers?.addEventListener("change", function() {
  userCheckboxes.forEach(cb => {
    cb.checked = this.checked;
    cb.closest(".user-row").classList.toggle("bg-blue-50/40", this.checked);
  });
  updateSelectionState();
});

btnClearSelection?.addEventListener("click", () => {
  userCheckboxes.forEach(cb => {
    cb.checked = false;
    cb.closest(".user-row").classList.remove("bg-blue-50/40");
  });
  updateSelectionState();
});

userSearchInput?.addEventListener("input", function() {
  const query = this.value.toLowerCase().trim();
  userRows.forEach(row => {
    const name = row.querySelector(".user-name")?.innerText.toLowerCase() || "";
    const email = row.querySelector(".user-email")?.innerText.toLowerCase() || "";
    row.classList.toggle("hidden", !(name.includes(query) || email.includes(query)));
  });
});
